==> mindhub.sql <==
CREATE DATABASE IF NOT EXISTS mindhub;
USE mindhub;

CREATE TABLE users (
  user_id INT AUTO_INCREMENT PRIMARY KEY,
  id_number VARCHAR(20) NOT NULL UNIQUE,
  first_name VARCHAR(50) NOT NULL,
  last_name VARCHAR(50) NOT NULL,
  email VARCHAR(100) NOT NULL UNIQUE,
  password_hash VARCHAR(255) NOT NULL,
  role ENUM("student", "counselor", "admin") NOT NULL DEFAULT "student",
  department_or_program VARCHAR(50) DEFAULT NULL,
  contact_number VARCHAR(15) DEFAULT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE counselor_schedules (
  schedule_id INT AUTO_INCREMENT PRIMARY KEY,
  counselor_id INT NOT NULL,
  day_of_week ENUM("Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday") NOT NULL,
  start_time TIME NOT NULL,
  end_time TIME NOT NULL,
  is_active TINYINT(1) DEFAULT 1,
  FOREIGN KEY (counselor_id) REFERENCES users(user_id)
);

CREATE TABLE appointments (
  appointment_id INT AUTO_INCREMENT PRIMARY KEY,
  student_id INT NOT NULL,
  counselor_id INT DEFAULT NULL,
  appointment_date DATE NOT NULL,
  start_time TIME NOT NULL,
  end_time TIME NOT NULL,
  nature_of_concern ENUM("Academic", "Personal", "Career", "Mental Health", "Other") NOT NULL,
  consultation_mode ENUM("Face-to-Face", "Online") NOT NULL DEFAULT "Face-to-Face",
  status ENUM("pending", "approved", "rejected", "completed", "cancelled") DEFAULT "pending",
  student_notes TEXT DEFAULT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (student_id) REFERENCES users(user_id),
  FOREIGN KEY (counselor_id) REFERENCES users(user_id)
);

CREATE TABLE counseling_notes (
  note_id INT AUTO_INCREMENT PRIMARY KEY,
  appointment_id INT NOT NULL UNIQUE,
  counselor_id INT NOT NULL,
  case_summary TEXT NOT NULL,
  action_taken TEXT DEFAULT NULL,
  follow_up_needed TINYINT(1) DEFAULT 0,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  FOREIGN KEY (appointment_id) REFERENCES appointments(appointment_id),
  FOREIGN KEY (counselor_id) REFERENCES users(user_id)
);

CREATE TABLE self_help_resources (
  resource_id INT AUTO_INCREMENT PRIMARY KEY,
  title VARCHAR(150) NOT NULL,
  category VARCHAR(50) NOT NULL,
  content TEXT NOT NULL,
  created_by INT NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (created_by) REFERENCES users(user_id)
);

<?php
session_start();
require_once 'config/db.php';

$error = '';
$success = '';

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $id_number  = trim($_POST['id_number']);
    $first_name = trim($_POST['first_name']);
    $last_name  = trim($_POST['last_name']);
    $email      = trim($_POST['email']);
    $program    = trim($_POST['department_or_program']);
    $contact    = trim($_POST['contact_number']);
    $password   = $_POST['password'];
    $confirm_pw = $_POST['confirm_password'];

    if (empty($id_number) || empty($first_name) || empty($last_name) || empty($email) || empty($password)) {
        $error = "Please fill in all required fields.";
    } elseif (!filter_var($email, FILTER_VALIDATE_EMAIL)) {
        $error = "Please enter a valid email address.";
    } elseif ($password !== $confirm_pw) {
        $error = "Passwords do not match.";
    } elseif (strlen($password) < 6) {
        $error = "Password must be at least 6 characters long.";
    } else {
        $checkStmt = $pdo->prepare("SELECT user_id FROM users WHERE id_number = ? OR email = ?");
        $checkStmt->execute([$id_number, $email]);

        if ($checkStmt->fetch()) {
            $error = "That Student ID Number or Email is already registered.";
        } else {
            $hashed_password = password_hash($password, PASSWORD_DEFAULT);

            $sql = "INSERT INTO users (
                        id_number, first_name, last_name, email, password_hash, role, department_or_program, contact_number
                    ) VALUES (?, ?, ?, ?, ?, 'student', ?, ?)";

            $insertStmt = $pdo->prepare($sql);

            $success_status = $insertStmt->execute([
                $id_number,
                $first_name,
                $last_name,
                $email,
                $hashed_password,
                $program,
                $contact
            ]);

            if ($success_status) {
                $success = "Registration successful! You may now proceed to log in.";
                $id_number = $first_name = $last_name = $email = $program = $contact = '';
            } else {
                $error = "Something went wrong while saving your account. Please try again.";
            }
        }
    }
}

            a.appointment_id,
            a.appointment_date,
            a.start_time,
            a.end_time,
            a.nature_of_concern,
            a.consultation_mode,
            a.status,
            a.created_at,
            u.first_name AS counselor_fname,
            u.last_name AS counselor_lname
        FROM appointments a
        LEFT JOIN users u ON a.counselor_id = u.user_id
        WHERE a.student_id = ?
        ORDER BY a.appointment_date DESC, a.start_time DESC";

    if ($apt['status'] === 'approved')  $approved_count++;

}
